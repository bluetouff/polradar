```sh
  -o candidats-presitracker.json \

printf '%s\n' '881c165df57fd3a3eb0da80d294c9ce89bef5ea18d948dcce74fe74c37e9d54c  candidats-presitracker.json' | shasum -a 256 -c -
```

Exécuter le code Python dans le même répertoire. Le rapprochement se fait sur les noms complets normalisés. Un homonyme dans le même fichier interrompt le calcul pour vérification.

```python
import json
import unicodedata
from collections import Counter
from pathlib import Path


def index(rows, display_name):
    result = {}
    for row in rows:
        name = display_name(row)
        decomposed = unicodedata.normalize("NFKD", name.casefold())
        key = "".join(char for char in decomposed if char.isascii() and char.isalnum())
        if not key or key in result:
            raise ValueError(f"Identité indéterminée ou plusieurs lignes pour {name}")
        result[key] = (name, row)
    return result


polradar = json.loads(Path("candidats-polradar.json").read_text())
presitracker = json.loads(Path("candidats-presitracker.json").read_text())
a = index(polradar, lambda row: row["name"])
b = index(presitracker["candidates"], lambda row: f'{row["firstName"]} {row["lastName"]}')
print("Communs", len(a.keys() & b.keys()))
print("Polradar uniquement", len(a.keys() - b.keys()))
print("PresiTracker2027 uniquement", Counter(b[k][1]["status"] for k in b.keys() - a.keys()))
for key in sorted(a.keys() | b.keys()):
    left_name, left = a.get(key, ("", {}))
    right_name, right = b.get(key, ("", {}))
    print(
        left_name or right_name,
        left.get("id", "·"),
        left.get("status", "·"),
        right.get("id", "·"),
        right.get("status", "·"),
        right.get("statusDate", "·"),
        sep="\t",
    )
```
